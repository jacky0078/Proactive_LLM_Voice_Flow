cmake_minimum_required(VERSION 3.16)
project(EdgeVoiceFlow VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
set(edge_build_testing "${BUILD_TESTING}")
include(FetchContent)
find_package(Threads REQUIRED)
option(EDGE_BUILD_COCKPIT "Build the proactive cockpit node" ON)
option(EDGE_BUILD_MANAGER "Build the TCP gateway" ON)
option(EDGE_BUILD_ASR "Build the real Sherpa-ONNX ASR node" ON)
option(EDGE_BUILD_TTS "Build the real SummerTTS node" ON)
option(EDGE_ENABLE_RKLLM "Link the real Rockchip LLM runtime" OFF)
set(VOICE_DEPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/voice-deps" CACHE PATH "Extracted voice dependencies")
set(RKLLM_ROOT "${VOICE_DEPS_ROOT}/rkllm" CACHE PATH "Matching RKLLM SDK root")
include(cmake/Dependencies.cmake)
add_subdirectory(infra-controller)
if(EDGE_BUILD_MANAGER OR EDGE_BUILD_COCKPIT)
  find_package(Boost REQUIRED)
  find_library(EDGE_GLOG_LIBRARY glog REQUIRED)
  add_subdirectory(network/src "${CMAKE_BINARY_DIR}/network")
endif()
if(EDGE_BUILD_MANAGER)
  add_subdirectory(unit-manager)
endif()
add_subdirectory(node)
# Sherpa's Eigen dependency changes the shared BUILD_TESTING cache entry.
# Preserve the caller's choice for Edge and for subsequent configurations.
set(BUILD_TESTING "${edge_build_testing}" CACHE BOOL "Build the testing tree" FORCE)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_subdirectory(tests)
endif()
