function(edge_voice_node name source)
    add_executable(edge_${name} common/main.cpp ${source})
    target_compile_definitions(edge_${name} PRIVATE EDGE_NODE_NAME="${name}")
    target_link_libraries(edge_${name} PRIVATE edge_voice_common)
    install(TARGETS edge_${name} RUNTIME DESTINATION bin)
endfunction()

if(EDGE_ENABLE_RKLLM)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        message(FATAL_ERROR "Real RKLLM runtime requires AArch64; no mock fallback is permitted")
    endif()
    find_path(RKLLM_INCLUDE_DIR rkllm.h PATHS "${RKLLM_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(RKLLM_LIBRARY rkllmrt PATHS "${RKLLM_ROOT}/aarch64" "${RKLLM_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    execute_process(COMMAND ${CMAKE_READELF} -h "${RKLLM_LIBRARY}" OUTPUT_VARIABLE rkllm_elf RESULT_VARIABLE rkllm_elf_rc)
    if(rkllm_elf_rc OR NOT rkllm_elf MATCHES "AArch64")
        message(FATAL_ERROR "RKLLM_LIBRARY is not an AArch64 ELF library")
    endif()
    target_sources(edge_llm PRIVATE llm/rkllm_backend.cpp)
    target_include_directories(edge_llm PRIVATE "${RKLLM_INCLUDE_DIR}")
    target_compile_definitions(edge_llm PRIVATE EDGE_HAS_RKLLM=1)
    target_link_libraries(edge_llm PRIVATE "${RKLLM_LIBRARY}")
    set_target_properties(edge_llm PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    install(FILES "${RKLLM_LIBRARY}" DESTINATION lib)
endif()
if(EDGE_BUILD_ASR)
    set(sherpa_root "${VOICE_DEPS_ROOT}/asr/sherpa-onnx")
    if(NOT EXISTS "${sherpa_root}/CMakeLists.txt")
        message(FATAL_ERROR "Missing ASR dependencies at ${sherpa_root}; run scripts/prepare_voice_deps.py first")
    endif()
    set(SHERPA_ONNX_BUILD_DIR "" CACHE PATH "Optional existing Sherpa-ONNX 1.11.3 shared-library build")
    if(SHERPA_ONNX_BUILD_DIR)
        add_library(edge_sherpa SHARED IMPORTED)
        set_target_properties(edge_sherpa PROPERTIES
            IMPORTED_LOCATION "${SHERPA_ONNX_BUILD_DIR}/lib/libsherpa-onnx-cxx-api.so"
            INTERFACE_INCLUDE_DIRECTORIES "${sherpa_root}")
    else()
        # Older dependencies must honor Sherpa's scoped static/shared library selection.
        set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
        set(BUILD_SHARED_LIBS ON)
        foreach(feature PORTAUDIO WEBSOCKET BINARY TTS SPEAKER_DIARIZATION PYTHON JNI GPU TESTS)
            set(SHERPA_ONNX_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
        endforeach()
        set(SHERPA_ONNX_BUILD_C_API_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(SHERPA_ONNX_ENABLE_C_API ON CACHE BOOL "" FORCE)
        add_subdirectory("${sherpa_root}" "${CMAKE_BINARY_DIR}/sherpa" EXCLUDE_FROM_ALL)
        add_library(edge_sherpa ALIAS sherpa-onnx-cxx-api)
    endif()
    edge_voice_node(asr asr/asr_task.cpp)
    target_include_directories(edge_asr PRIVATE "${sherpa_root}")
    target_link_libraries(edge_asr PRIVATE edge_sherpa)
endif()
